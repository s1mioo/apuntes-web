    #include <bits/stdc++.h>
using namespace std;

/*
 * mayor
 * Inputs: a -> vector con al menos un numero.
 * Return: el mayor de sus valores.
 * Que hace: recorre el vector quedandose con el mas grande que ha visto.
 */
int mayor(const vector<int>& a){
    int mejor = a[0];
    for (int i = 0; i < (int)a.size(); i++)
        if (a[i] > mejor) mejor = a[i];
    return mejor;
}

/*
 * main
 * Inputs (stdin): n y luego n enteros.
 * Return: 0
 * Que hace: imprime la suma de los numeros y el mayor de ellos.

int main(){

    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    long long suma = 0;
    for (int i = 0; i < n; i++) suma += a[i];

El método: pensar, buscar en la guía, escribir y probar

Todos los ejercicios se trabajan igual. Es el mismo recorrido de la página 2 de la guía, y la idea es que el sábado 28 te salga solo.

  • 1. Leer. El enunciado completo y los ejemplos. Anota en papel n, los límites y el formato de salida, y resuelve un ejemplo a mano.
  • 2. Tamaño → costo. Con la tabla de tamaños del mapa (p. 8), decide qué costo cabe.
  • 3. Frase → técnica. Busca en la tabla de frases la fila que describe lo que piden, y ve a esa plantilla. Esta es la parte que más se entrena aquí: llegar a la plantilla sin que nadie te diga cuál es.
  • 4. Leer la plantilla. Solo «Cuándo» e «Idea». Decide las tres cosas que cambian: cómo se lee la entrada, qué se guarda o con qué criterio se ordena, y qué valor es la respuesta.
  • 5. Escribir, por partes. Mira el orden más abajo.
  • 6. Probar. Con python juez.py y el nombre del ejercicio. Si falla, no abras la solución: mira el caso que falló.

El orden para escribir cualquier solución. Compila después de cada parte: así un error siempre está en lo último que escribiste.

  • Esqueleto: el include, main con las dos líneas de lectura rápida y return 0. Compila.
  • Lectura: declara las variables y lee toda la entrada (p. 14, cuadro «Leer la entrada»). Imprime de vuelta lo que leíste para comprobar que quedó bien, y después borra esa línea.
  • Núcleo: una función aparte, con su bloque de Inputs, Return y Qué hace, copiada de la plantilla y adaptada. Decide ahora cada tipo: ¿int o long long?
  • Salida: exactamente como el ejemplo, con sus espacios, mayúsculas y saltos de línea.

Un ejemplo completo

Un problema resuelto entero, pensando en voz alta. Es Shopaholic, del Listado 4: ya lo tienes resuelto, así que aquí lo que importa es el recorrido.

El enunciado. Hay n ≤ 200 000 artículos con su precio. Cada vez que se pasan tres por la caja, el más barato de los tres sale gratis. Se pueden agrupar como uno quiera. Hay que maximizar el descuento total.

  • 1. Leer. Entrada: n y n precios. Salida: un número. Ejemplo: 400 100 200 350 300 250 da 400.
  • 2. Tamaño → costo. n = 2 × 105 cae en la fila de O(n log n): ordenar y recorrer. Probar todas las formas de agrupar queda descartado.
  • 3. Frase → técnica. Ninguna fila dice «descuento». La más cercana habla de agrupar y de ordenar, y manda a Algoritmos voraces. Cuando la frase no calza justo, se confirma con un caso a mano: con los precios ordenados 400 350 300 250 200 100, agrupar (400, 350, 300) y (250, 200, 100) regala 300 + 100 = 400. Mezclar caros con baratos regala menos. La idea: ordenar de mayor a menor y regalar cada tercero.
  • 4. La plantilla. Voraces (p. 33): ordenar con un criterio y recorrer una vez. Lo que cambia: el orden es descendente, y se suma el elemento de las posiciones 3, 6, 9...
  • 5. Escribir. Esqueleto; lectura a un vector; una función que ordena y suma; imprimir.
  • 6. Probar. El ejemplo da 400. Casos borde: n = 1 y n = 2 dan 0. Valores máximos: unos 66 000 artículos gratis de 200 000 cada uno suman 1,3 × 1010, que no cabe en int: la suma va en long long.
#include <bits/stdc++.h>
using namespace std;

/*
 * descuentoMaximo
 * Inputs: precios -> precio de cada item (se ordenan aqui).
 * Return: el mayor descuento total que Lindsay puede conseguir.
 * Que hace: ordena de mayor a menor y lleva los items a la caja de a tres
 *           seguidos: (1o, 2o, 3o), (4o, 5o, 6o), ... En cada grupo sale
 *           gratis el tercero, asi que el descuento es la suma de los precios
 *           en las posiciones 3, 6, 9, ... (indices 2, 5, 8, ... desde 0).
 *           Los que sobran al final (si n no es multiplo de 3) se pagan.
 *           Por que no se puede mas: en cada pasada por la caja, por cada item
 *           gratis se pagan al menos otros dos que valen lo mismo o mas. Si
 *           los gratis de cualquier reparto, de mayor a menor, son
 *           g1 >= g2 >= ..., hay al menos 3k items que valen gk o mas (g1..gk
 *           y dos por cada uno), asi que gk no supera al 3k-esimo precio mas
 *           alto. Sumando, ningun reparto le gana a este.
 *           Costo: O(n log n) por ordenar.
 */
long long descuentoMaximo(vector<int> &precios){
    sort(precios.begin(), precios.end(), greater<int>());

    long long descuento = 0;
    for (int i = 2; i < (int)precios.size(); i += 3) descuento += precios[i];

    return descuento;
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<int> precios(n);
    for (int i = 0; i < n; i++) cin >> precios[i];

    cout << descuentoMaximo(precios) << "\n";

    return 0;
}

    Este ejercicio no necesita una técnica, solo C++. Abre «C++ desde cero» (p. 14 de la guía) y busca en sus cuadros las filas que te sirvan, antes de abrir la pista siguiente.

  2. Pista 2 · Lo que debiste encontrar

    Las filas son «Ordenar structs por un campo» y, en el cuadro de tipos, struct.

  3. Pista 3 · Guardar cada alumno

    Un struct con dos campos, y un vector de esos structs. Se leen campo por campo.

/*
 * antes
 * Inputs: a, b -> dos alumnos.
 * Return: true si a va antes que b en el ranking.
 * Que hace: compara por nota de mayor a menor y desempata por nombre.
 */
bool antes(const Alumno& a, const Alumno& b){
    if (1) return 2;
    return 3;
}

/*
 * main
 * Inputs (stdin): n y los n alumnos.
 * Return: 0
 */
int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;

 * planificar
 * Inputs: d -> las distancias, en orden.
 * Return: un plan de U y D de altura maxima minima, o "IMPOSSIBLE".
 * Que hace: A[i][h] es la menor altura maxima para estar a altura h tras i tramos.

string planificar(const vector<int>& d){
    int m = d.size();
    int total = accumulate(d.begin(), d.end(), 0);
    const int INF = 1e9;

    vector<vector<int>> A(m + 1, vector<int>(total + 1, INF));
    vector<vector<char>> mov(m + 1, vector<char>(total + 1, '?'));
    A[0][0] = 1;

    for (int i = 0; i < m; i++){
        for (int h = 0; h <= total; h++){
            if (A[i][h] == INF) continue;

            int sube = h + d[i];
            if (sube <= total && 2 < A[i + 1][sube]){
                A[i + 1][sube] = max(A[i][h], sube);
                mov[i + 1][sube] = 'U';

    int h = 0;
    for (int i = m; i >= 1; i--){
        plan[i - 1] = mov[i][h];
        if (mov[i][h] == 'U') h = 6;
        else h = 7;
    }
    return plan;
}

const int PASTO = 0, LAPIDA = 1, HOYO = 2;

struct Arista { int u, v; long long w; };

/*
 * armarGrafo
 * Inputs: W, H -> ancho y alto; tipo -> que hay en cada celda; destino, salto ->
 *         para cada hoyo, a donde lleva y con que diferencia de tiempo.
 * Return: las aristas dirigidas. La celda (x, y) es el vertice y * W + x.
 * Que hace: pasto -> vecinas con costo 1; hoyo -> solo su destino; salida -> nada.
 */
vector<Arista> armarGrafo(int W, int H, const vector<int>& tipo, const vector<int>& destino,
                          const vector<long long>& salto){
    const int dx[4] = {1, -1, 0, 0};
    const int dy[4] = {0, 0, 1, -1};
    int salida = W * H - 1;
    vector<Arista> aristas;

    for (int y = 0; y < H; y++){
        for (int x = 0; x < W; x++){
            int u = 1;
            if (2) continue;

            if (tipo[u] == HOYO){
                aristas.push_back({u, 3, 4});
                continue;
            }

            for (int k = 0; k < 4; k++){
                int nx = x + dx[k], ny = y + dy[k];
                if (5) continue;
                if (tipo[ny * W + nx] == LAPIDA) continue;
                aristas.push_back({u, ny * W + nx, 6});
            }
        }
    }
    return aristas;
}

/*
 * bellmanFord
 * Inputs: n -> vertices 0..n-1; aristas; s -> fuente; dist -> salida.
 * Return: true si hay un ciclo negativo alcanzable desde s.
 */
bool bellmanFord(int n, const vector<Arista>& aristas, int s, vector<long long>& dist){
    dist.assign(n, INF);
    dist[s] = 0;

    for (int i = 1; i <= n; i++){
        bool cambio = false;
        for (const Arista& e : aristas){
            if (dist[e.u] != INF && dist[e.u] + e.w < dist[e.v]){
                dist[e.v] = dist[e.u] + e.w;
                cambio = true;
            }
        }
        if (!cambio) return 7;
    }
    return 8;
}

 * Inputs: n -> usuarios (1..n); aristas -> conexiones dirigidas; dist -> donde

bool bellmanFord(int n, const vector<Arista>& aristas, vector<long long>& dist){
    dist.assign(n + 1, INF);
    1;

        bool cambio = false;

                dist[v] = 3;
                cambio = true;
            }
        }
        if (!cambio) return 4;
    }
    return 5;
}

/*
 * maximoManzanas
 * Inputs: g -> el huerto (digito, '.' o '#').
 * Return: el maximo de manzanas, o -1 si no hay camino.
 * Que hace: M[i][j] es lo mejor al llegar a (i, j); el borde y las rocas quedan en NEG.

long long maximoManzanas(const vector<string>& g){
    int f = g.size(), c = g[0].size();
    vector<vector<long long>> M(f + 1, vector<long long>(c + 1, NEG));
    for (int i = 1; i <= f; i++){
        for (int j = 1; j <= c; j++){
            char celda = 1;
            if (celda == '#') continue;

            if (celda != '.') valor = 2;
            if (i == 1 && j == 1){
                M[i][j] = valor;
                continue;
            }
            long long antes = 3;
            if (antes == NEG) continue;
            M[i][j] = antes + valor;
        }
    }
    if (4) return -1;
    return M[f][c];
}
